cmake_minimum_required(VERSION 3.24)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_TRY_COMPILE_CONFIGURATION Release)
project(ColdEditRelink LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(EMSCRIPTEN)
  add_compile_options(-pthread -fexceptions)
  add_link_options(-pthread -fexceptions --use-port=contrib.glfw3)
  file(GLOB objects objects/*.o)
  set(suffix .a)
elseif(MSVC)
  file(GLOB objects objects/*.obj)
  set(suffix .lib)
else()
  file(GLOB objects objects/*.o)
  set(suffix .a)
endif()
if(NOT objects)
  message(FATAL_ERROR "Use the matching platform/compiler for this kit; see TOOLCHAIN.txt.")
endif()
set(lame_SOURCE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/lame)
include(Lame.cmake)
set_source_files_properties(${objects} PROPERTIES EXTERNAL_OBJECT TRUE)
add_executable(coldedit ${objects})
set_target_properties(coldedit PROPERTIES LINKER_LANGUAGE CXX)
foreach(name coldedit_audio coldedit_imgui coldedit_core FLAC)
  target_link_libraries(coldedit PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/lib/${name}${suffix})
endforeach()
target_link_libraries(coldedit PRIVATE coldedit_lame)
if(EMSCRIPTEN)
  set_target_properties(coldedit PROPERTIES SUFFIX .js)
  target_link_options(coldedit PRIVATE
    -sMIN_WEBGL_VERSION=2 -sMAX_WEBGL_VERSION=2 -sAUDIO_WORKLET=1 -sWASM_WORKERS=1 -sASYNCIFY
    -sASYNCIFY_STACK_SIZE=131072 -sSTACK_SIZE=1048576 -sPTHREAD_POOL_SIZE=4 -sPTHREAD_POOL_SIZE_STRICT=1
    -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=134217728 -sMAXIMUM_MEMORY=536870912
    -sENVIRONMENT=web,worker -sFORCE_FILESYSTEM=1 "-sEXPORTED_RUNTIME_METHODS=['ccall','FS']"
    "--preload-file=${CMAKE_CURRENT_SOURCE_DIR}/fonts/Roboto-Medium.ttf@/fonts/Roboto-Medium.ttf"
    "--preload-file=${CMAKE_CURRENT_SOURCE_DIR}/fonts/Cousine-Regular.ttf@/fonts/Cousine-Regular.ttf")
elseif(WIN32)
  set_target_properties(coldedit PROPERTIES WIN32_EXECUTABLE TRUE)
  target_link_options(coldedit PRIVATE /ENTRY:mainCRTStartup)
  target_link_libraries(coldedit PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/lib/glfw.lib comdlg32 ole32 psapi shell32 opengl32)
else()
  find_package(Threads REQUIRED)
  find_package(OpenGL REQUIRED)
  target_link_libraries(coldedit PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/lib/glfw.a Threads::Threads OpenGL::GL ${CMAKE_DL_LIBS} m)
endif()
