cmake_minimum_required(VERSION 3.24)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(ColdEdit VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(COLDEDIT_BUILD_APP "Build the native desktop application" ON)
option(COLDEDIT_SANITIZE "Enable address/undefined behavior sanitizers (Clang/GCC)" OFF)
if(EMSCRIPTEN)
  # The contrib port supports GLFW 3.4, browser DPI and proper pointer handling.
  add_compile_options(-pthread -fexceptions --use-port=contrib.glfw3)
  add_link_options(-pthread -fexceptions --use-port=contrib.glfw3)
endif()
find_package(Threads REQUIRED)
include(CTest)
include(cmake/Codecs.cmake)
add_library(coldedit_core src/audio_file.cpp src/document.cpp src/processing.cpp src/codecs.cpp src/dr_mp3.c)
target_include_directories(coldedit_core PUBLIC src)
target_include_directories(coldedit_core PRIVATE ${dr_libs_SOURCE_DIR})
target_link_libraries(coldedit_core PUBLIC Threads::Threads PRIVATE FLAC coldedit_lame)
if(MSVC)
  target_compile_options(coldedit_core PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(coldedit_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(COLDEDIT_SANITIZE AND NOT MSVC)
  target_compile_options(coldedit_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(coldedit_core PUBLIC -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
  add_executable(coldedit_tests tests/core_tests.cpp)
  target_link_libraries(coldedit_tests PRIVATE coldedit_core)
  if(MSVC)
    target_compile_options(coldedit_tests PRIVATE /utf-8)
  endif()
  add_test(NAME core COMMAND coldedit_tests)
  add_executable(coldedit_codec_tests tests/codec_tests.cpp)
  target_link_libraries(coldedit_codec_tests PRIVATE coldedit_core)
  target_compile_definitions(coldedit_codec_tests PRIVATE "COLDEDIT_FIXTURES=\"${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures\"")
  add_test(NAME codecs COMMAND coldedit_codec_tests)
  add_executable(coldedit_benchmark tests/benchmark.cpp)
  target_link_libraries(coldedit_benchmark PRIVATE coldedit_core)
  if(WIN32)
    target_link_libraries(coldedit_benchmark PRIVATE psapi)
  endif()
endif()
if(COLDEDIT_BUILD_APP)
  include(FetchContent)
  function(coldedit_dependency name url hash)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${name}.tar.gz")
      set(url "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${name}.tar.gz")
    endif()
    if(name STREQUAL "glfw")
      FetchContent_Declare(${name} URL "${url}" URL_HASH "SHA256=${hash}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    else()
      FetchContent_Declare(${name} URL "${url}" URL_HASH "SHA256=${hash}" DOWNLOAD_EXTRACT_TIMESTAMP TRUE SOURCE_SUBDIR coldedit-no-cmake)
    endif()
  endfunction()
  coldedit_dependency(imgui https://github.com/ocornut/imgui/archive/refs/tags/v1.91.9b.tar.gz 8e1bbc76c71d74fef2fb85db7e7ca8eba13d6a86623c54992b60162db554ffdb)
  coldedit_dependency(glfw https://github.com/glfw/glfw/archive/refs/tags/3.4.tar.gz c038d34200234d071fae9345bc455e4a8f2f544ab60150765d7704e08f3dac01)
  coldedit_dependency(miniaudio https://github.com/mackron/miniaudio/archive/refs/tags/0.11.23.tar.gz 1062155cc5882b55c48cb37f57a4dc783669e83ae0838535c62b206eeb1587a6)
  set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
  set(MINIAUDIO_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(MINIAUDIO_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  # Populate header/source-only dependencies without adopting their build systems.
  FetchContent_MakeAvailable(imgui miniaudio)
  if(NOT EMSCRIPTEN)
    FetchContent_MakeAvailable(glfw)
    find_package(OpenGL REQUIRED)
  endif()
  set(coldedit_glfw_backend ${imgui_SOURCE_DIR}/backends/imgui_impl_glfw.cpp)
  if(EMSCRIPTEN)
    include(cmake/WebImGui.cmake)
  endif()
  add_library(coldedit_imgui STATIC
    ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
    ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
    ${coldedit_glfw_backend}
    ${imgui_SOURCE_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(coldedit_imgui PUBLIC ${imgui_SOURCE_DIR} ${imgui_SOURCE_DIR}/backends)
  # Stereo pixel columns with antialiased ends can exceed 64K vertices at high
  # DPI. WebGL lacks base-vertex draws; use 32-bit indices on both targets.
  target_compile_definitions(coldedit_imgui PUBLIC "ImDrawIdx=unsigned int")
  if(EMSCRIPTEN)
    target_compile_definitions(coldedit_imgui PUBLIC IMGUI_IMPL_OPENGL_ES3)
  else()
    target_link_libraries(coldedit_imgui PUBLIC glfw OpenGL::GL)
  endif()
  add_library(coldedit_audio STATIC src/playback.cpp)
  target_include_directories(coldedit_audio PRIVATE ${miniaudio_SOURCE_DIR})
  target_link_libraries(coldedit_audio PUBLIC coldedit_core ${CMAKE_DL_LIBS})
  if(EMSCRIPTEN)
    target_compile_definitions(coldedit_audio PRIVATE MA_ENABLE_AUDIO_WORKLETS)
  endif()
  if(WIN32)
    target_link_libraries(coldedit_audio PUBLIC ole32)
    target_compile_definitions(coldedit_audio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  endif()
  if(BUILD_TESTING)
    add_executable(coldedit_playback_tests tests/playback_tests.cpp)
    target_link_libraries(coldedit_playback_tests PRIVATE coldedit_audio)
    add_test(NAME playback COMMAND coldedit_playback_tests)
  endif()
  if(EMSCRIPTEN)
    add_executable(coldedit src/main_web.cpp src/editor.cpp src/platform.cpp)
    set_target_properties(coldedit PROPERTIES SUFFIX ".js" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/site")
    target_link_options(coldedit PRIVATE
      -sMIN_WEBGL_VERSION=2 -sMAX_WEBGL_VERSION=2
      -sAUDIO_WORKLET=1 -sWASM_WORKERS=1 -sASYNCIFY
      -sASYNCIFY_STACK_SIZE=131072 -sSTACK_SIZE=1048576
      -sPTHREAD_POOL_SIZE=4 -sPTHREAD_POOL_SIZE_STRICT=1
      -sALLOW_MEMORY_GROWTH=1 -sINITIAL_MEMORY=134217728 -sMAXIMUM_MEMORY=536870912
      -sENVIRONMENT=web,worker -sFORCE_FILESYSTEM=1
      "-sEXPORTED_RUNTIME_METHODS=['ccall','FS']"
      "SHELL:--preload-file \"${imgui_SOURCE_DIR}/misc/fonts/Roboto-Medium.ttf@/fonts/Roboto-Medium.ttf\""
      "SHELL:--preload-file \"${imgui_SOURCE_DIR}/misc/fonts/Cousine-Regular.ttf@/fonts/Cousine-Regular.ttf\"")
    foreach(asset index.html app.js style.css _headers)
      configure_file("web/${asset}" "${CMAKE_BINARY_DIR}/site/${asset}" COPYONLY)
    endforeach()
    install(DIRECTORY "${CMAKE_BINARY_DIR}/site/" DESTINATION .)
    install(FILES web/serve.mjs DESTINATION .)
    install(DIRECTORY "web/licenses/" DESTINATION licenses)
  else()
    add_executable(coldedit src/main.cpp src/editor.cpp src/platform.cpp)
    install(TARGETS coldedit RUNTIME DESTINATION .)
  endif()
  target_link_libraries(coldedit PRIVATE coldedit_audio coldedit_imgui)
  if(WIN32)
    target_link_libraries(coldedit PRIVATE comdlg32 ole32 psapi shell32)
    target_compile_definitions(coldedit PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    set_target_properties(coldedit PROPERTIES WIN32_EXECUTABLE TRUE)
    target_link_options(coldedit PRIVATE /ENTRY:mainCRTStartup)
  endif()
  if(MSVC)
    target_compile_options(coldedit PRIVATE /W4 /utf-8)
  endif()
  install(FILES README.md PERFORMANCE.md WEB.md DESTINATION .)
  install(FILES ${imgui_SOURCE_DIR}/LICENSE.txt DESTINATION licenses RENAME Dear-ImGui.txt)
  if(NOT EMSCRIPTEN)
    install(FILES ${glfw_SOURCE_DIR}/LICENSE.md DESTINATION licenses RENAME GLFW.txt)
  endif()
  install(FILES ${miniaudio_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME miniaudio.txt)
  install(FILES ${flac_SOURCE_DIR}/COPYING.Xiph DESTINATION licenses RENAME libFLAC.txt)
  install(FILES ${dr_libs_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME dr_libs.txt)
  install(FILES ${lame_SOURCE_DIR}/COPYING DESTINATION licenses RENAME LAME.txt)
  include(cmake/Relink.cmake)
endif()
